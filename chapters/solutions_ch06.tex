\section{Exercise Solutions}

\begin{solution}
The Jacobi triple product identity $\sum_{n=-\infty}^{\infty} z^n q^{n^2} = \prod_{m=1}^{\infty}(1-q^{2m})(1+zq^{2m-1})(1+z^{-1}q^{2m-1})$ is proved by showing both sides satisfy the same functional equation and have the same zeros. Setting $z = 1$ gives $\vartheta_3(0|\tau) = \prod_{m=1}^{\infty}(1-q^{2m})(1+q^{2m-1})^2$.
\end{solution}

\begin{solution}
The modular group $SL(2,\Z)$ is generated by $S = \begin{pmatrix} 0 & -1 \\ 1 & 0 \end{pmatrix}$ and $T = \begin{pmatrix} 1 & 1 \\ 0 & 1 \end{pmatrix}$. The theta function\index{theta function} transforms as $\vartheta_3(0|-1/\tau) = \sqrt{-i\tau}\,\vartheta_3(0|\tau)$ and $\vartheta_3(0|\tau+1) = e^{\pi i/4}\vartheta_3(0|\tau)$. These generate the full transformation law under $\Gamma(2)$.
\end{solution}

\begin{solution}
For a quadratic form $Q$ of rank $n$, the theta series $\Theta_Q(z) = \sum_{x \in \Z^n} e^{\pi i Q(x) z}$ is a modular form\index{modular form} of weight $n/2$ on $\Gamma_0(4)$ with character. For $Q(x) = x_1^2 + \cdots + x_n^2$, we get $\Theta_Q(z) = \vartheta_3(0|z)^n$, which counts representations of integers as sums of $n$ squares.
\end{solution}

\begin{solution}
The Dedekind eta function\index{Dedekind eta function} function $\eta(\tau) = e^{\pi i \tau/12}\prod_{n=1}^{\infty}(1-e^{2\pi i n\tau})$ satisfies $\eta(-1/\tau) = \sqrt{-i\tau}\,\eta(\tau)$ and $\eta(\tau+1) = e^{\pi i/12}\eta(\tau)$. It is a modular form\index{modular form} of weight $1/2$ with a multiplier system. The Weber functions $\mathfrak{f}, \mathfrak{f}_1, \mathfrak{f}_2$ are defined in terms of eta and generate cyclotomic fields.
\end{solution}
