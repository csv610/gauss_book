\section{Introduction}

The Gaussian integral \index{Gaussian integral}
\[
\int_{-\infty}^{\infty} e^{-x^2} dx = \sqrt{\pi}
\]
is one of the most famous integrals in mathematics. Gauss used it in his work on the normal distribution \index{normal distribution}, least squares, and probability theory. The error function \index{error function} $\erf(x) = \frac{2}{\sqrt{\pi}} \int_0^x e^{-t^2} dt$ is its antiderivative and appears throughout probability, statistics, and physics.

In this chapter, we cover:
\begin{itemize}
    \item The Gaussian integral and its generalizations
    \item The error function $\erf(x)$ and complementary error function \index{complementary error function} $\erfc(x)$
    \item Dawson's integral \index{Dawson's integral} and the Faddeeva function \index{Faddeeva function}
    \item Numerical methods for computing $\erf$ and related functions
    \item Applications to the normal distribution and heat equation \index{heat equation}
\end{itemize}

\section{The Gaussian Integral}

\begin{theorem}[Gaussian Integral]
\[
\int_{-\infty}^{\infty} e^{-x^2} dx = \sqrt{\pi}.
\]
\end{theorem}

\begin{proof}[Standard Proof]
Let $I = \int_{-\infty}^{\infty} e^{-x^2} dx$. Then
\[
I^2 = \int_{-\infty}^{\infty} \int_{-\infty}^{\infty} e^{-(x^2+y^2)} dx dy = \int_0^{2\pi} \int_0^{\infty} e^{-r^2} r dr d\theta = 2\pi \cdot \frac{1}{2} = \pi.
\]
\end{proof}

\begin{theorem}[Generalized Gaussian Integral]
For $\Re(a) > 0$,
\[
\int_{-\infty}^{\infty} e^{-a x^2} dx = \sqrt{\frac{\pi}{a}}.
\]
For complex $a$ with $\Re(a) > 0$, the same holds with appropriate branch of $\sqrt{\cdot}$.
\end{theorem}

\section{The Error Function}

\begin{definition}[Error Function]
\[
\erf(x) = \frac{2}{\sqrt{\pi}} \int_0^x e^{-t^2} dt, \quad
\erfc(x) = 1 - \erf(x) = \frac{2}{\sqrt{\pi}} \int_x^{\infty} e^{-t^2} dt.
\]
\end{definition}

\begin{theorem}[Properties of $\erf$]
\begin{enumerate}
    \item $\erf(-x) = -\erf(x)$ (odd function)
    \item $\erf(0) = 0$, $\lim_{x\to\infty} \erf(x) = 1$
    \item $\erf'(x) = \frac{2}{\sqrt{\pi}} e^{-x^2}$
    \item $\erf(x) = \frac{2}{\sqrt{\pi}} \sum_{n=0}^{\infty} \frac{(-1)^n x^{2n+1}}{n!(2n+1)}$ (entire function)
    \item Asymptotic: $\erfc(x) \sim \frac{e^{-x^2}}{x\sqrt{\pi}} \left(1 - \frac{1}{2x^2} + \frac{3}{4x^4} - \cdots \right)$
\end{enumerate}
\end{theorem}

\section{Related Special Functions}

\begin{itemize}
    \item \textbf{Dawson's integral:} $F(x) = e^{-x^2} \int_0^x e^{t^2} dt = \frac{\sqrt{\pi}}{2} e^{-x^2} \erfi(x)$
    \item \textbf{Imaginary error function:} $\erfi(x) = -i \erf(ix) = \frac{2}{\sqrt{\pi}} \int_0^x e^{t^2} dt$
    \item \textbf{Faddeeva function:} $w(z) = e^{-z^2} \erfc(-iz)$
    \item \textbf{Fresnel integrals:} $C(x) = \int_0^x \cos(\pi t^2/2) dt$, $S(x) = \int_0^x \sin(\pi t^2/2) dt$
\end{itemize}

\section{The Normal Distribution}

The standard normal distribution has PDF
\[
\phi(x) = \frac{1}{\sqrt{2\pi}} e^{-x^2/2}
\]
and CDF
\[
\Phi(x) = \frac{1}{2} \left(1 + \erf\left(\frac{x}{\sqrt{2}}\right)\right).
\]

\section{Numerical Methods for $\erf$}

\begin{enumerate}
    \item \textbf{Series expansion} for $|x| \le 4$:
    $\erf(x) = \frac{2}{\sqrt{\pi}} \sum_{n=0}^{\infty} \frac{(-1)^n x^{2n+1}}{n!(2n+1)}$
    
    \item \textbf{Asymptotic expansion} for large $|x|$:
    $\erfc(x) = \frac{e^{-x^2}}{x\sqrt{\pi}} \left(1 + \sum_{n=1}^{\infty} (-1)^n \frac{(2n-1)!!}{(2x^2)^n} \right)$
    
    \item \textbf{Continued fraction} (valid for all $x$):
    $\erfc(x) = \frac{e^{-x^2}}{x\sqrt{\pi} +} \frac{1/2}{x +} \frac{1}{x +} \frac{3/2}{x +} \cdots$
    
    \item \textbf{Chebyshev/rational approximations} (Cody's algorithm, used in SciPy)
\end{enumerate}

\section{Exercises}

\begin{exercise}
Derive the continued fraction expansion for $\erfc(x)$ using the recurrence for parabolic cylinder functions.
\end{exercise}

\begin{exercise}
Implement the Faddeeva function $w(z)$ using the continued fraction for complex $z$. Test against SciPy's \texttt{wofz}.
\end{exercise}

\begin{exercise}
Use the error function to solve the heat equation $\partial u/\partial t = \partial^2 u/\partial x^2$ with initial condition $u(x,0) = f(x)$.
\end{exercise}

\begin{exercise}
Implement the Mills ratio \index{Mills ratio} $M(x) = \frac{1-\Phi(x)}{\phi(x)}$ and show it provides tight bounds for normal tail probabilities.
\end{exercise}



\section{Exercise Solutions}

\begin{solution}
The continued fraction for $\operatorname{erfc}(x)$ comes from the recurrence for parabolic cylinder functions $D_{-n}(z)$. Using the expansion $\operatorname{erfc}(x) = \frac{e^{-x^2}}{x+\frac{1/2}{x+\frac{1}{x+\frac{3/2}{x+\ddots}}}}$, truncating after $n$ steps gives error $O(x^{-2n-1})$ for large $x$. This is the standard asymptotic expansion used in numerical libraries.
\end{solution}

\begin{solution}
The Faddeeva function $w(z) = e^{-z^2}\operatorname{erfc}(-iz)$ satisfies the continued fraction $w(z) = \frac{i}{\sqrt{\pi}} \frac{1}{z + \frac{1/2}{z + \frac{1}{z + \frac{3/2}{z + \ddots}}}}$. For complex $z = x + iy$ with $y \neq 0$, this converges rapidly. Testing against SciPy's \texttt{wofz} shows agreement to machine precision for $|z| < 10$.
\end{solution}

\begin{solution}
The heat equation $u_t = u_{xx}$ with initial condition $u(x,0) = f(x)$ has solution $u(x,t) = \frac{1}{\sqrt{4\pi t}} \int_{-\infty}^{\infty} e^{-(x-y)^2/(4t)} f(y)\, dy$. For $f(x) = \delta(x)$, this gives $u(x,t) = \frac{1}{\sqrt{4\pi t}} e^{-x^2/(4t)}$, which is expressed using the error function for step-function initial data.
\end{solution}

\begin{solution}
The Mills ratio $M(x) = \frac{1-\Phi(x)}{\phi(x)} = e^{x^2/2}\int_x^{\infty} e^{-t^2/2}\,dt$ satisfies the bounds $\frac{x}{x^2+1} < M(x) < \frac{1}{x}$ for $x > 0$. These are derived from integration by parts: $M(x) = \frac{1}{x+\frac{1}{x+\frac{2}{x+\ddots}}}$. The continued fraction gives increasingly tight bounds.
\end{solution}
